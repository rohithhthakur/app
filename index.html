import pandas as pd # Load the CSV files df1 = pd.read_csv('path/to/your/first_file.csv') df2 = pd.read_csv('path/to/your/second_file.csv') # Get data types for each column dtypes_df1 = df1.dtypes dtypes_df2 = df2.dtypes # Compare data types and find differences differences = {} for column in dtypes_df1.index: if column in dtypes_df2: if dtypes_df1[column] != dtypes_df2[column]: differences[column] = {'First CSV Type': dtypes_df1[column], 'Second CSV Type': dtypes_df2[column]} # Print the differences if differences: print("Columns with different schemas:") for column, types in differences.items(): print(f"Column: {column}, First CSV Type: {types['First CSV Type']}, Second CSV Type: {types['Second CSV Type']}") else: print("All columns have the same data types.")
