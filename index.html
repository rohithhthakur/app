import boto3 # Initialize a session using Amazon CloudWatch client = boto3.client('logs') # Define your log group and log stream log_group_name = 'your-log-group-name' log_stream_name = 'your-log-stream-name' # Fetch the logs response = client.get_log_events( logGroupName=log_group_name, logStreamName=log_stream_name, startFromHead=True # Start from the beginning of the log stream ) # Write logs to a text file with open('log_output.txt', 'w') as file: for event in response['events']: file.write(event['message'] + '\n') print("Logs have been written to log_output.txt")
